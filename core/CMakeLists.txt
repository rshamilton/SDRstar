cmake_minimum_required(VERSION 3.13)
project(sdrstar_core CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
if (NOT CMAKE_BUILD_TYPE)
    set(CMAKE_BUILD_TYPE Release)
endif ()

file(GLOB SDRSTAR_CORE_SRC "src/*.cpp")
add_library(sdrstar_core STATIC ${SDRSTAR_CORE_SRC})
target_include_directories(sdrstar_core PUBLIC include)
target_include_directories(sdrstar_core PRIVATE third_party)
target_compile_options(sdrstar_core PRIVATE -Wall -Wextra -Wno-unused-parameter)
find_package(Threads REQUIRED)
target_link_libraries(sdrstar_core PUBLIC Threads::Threads)

option(SDRSTAR_BUILD_TESTS "Build sdrstar_core unit tests" ON)
if (SDRSTAR_BUILD_TESTS)
    enable_testing()
    file(GLOB SDRSTAR_TEST_SRC "tests/test_*.cpp")
    foreach (src ${SDRSTAR_TEST_SRC})
        get_filename_component(name ${src} NAME_WE)
        add_executable(${name} ${src})
        target_link_libraries(${name} PRIVATE sdrstar_core)
        target_include_directories(${name} PRIVATE tests)
        add_test(NAME ${name} COMMAND ${name})
    endforeach ()
endif ()
